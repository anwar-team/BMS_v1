```php
public function mount(int $bookId, ?int $pageNumber = null)
{
    $this->bookId = $bookId;
    $this->pageNumber = $this->coercePage($pageNumber);
    $this->loadToc(); // Preload TOC from cache or DB
    $this->loadPage(); // Load the current page
}

public function loadToc(): void
{
    // Fetch volumes and chapters. Cache results to improve performance.
    $this->toc = $this->book->volumes()->with('chapters')->get();
}

public function loadPage(): void
{
    // Load current page data from DB
    $this->currentPage = $this->book->pages()->where('page_number', $this->pageNumber)->first();
    
    // If there's a source URL, use `html_content`, otherwise fallback to `content`.
    $this->currentContent = $this->currentPage->source_url ? $this->currentPage->html_content : $this->currentPage->content;
    
    $this->navigation = [
        'total_pages' => $this->book->pages()->count(),
        'prev' => $this->getPrevPage(),
        'next' => $this->getNextPage(),
    ];
}

public function getPrevPage()
{
    return $this->pageNumber > 1 ? $this->pageNumber - 1 : null;
}

public function getNextPage()
{
    return $this->pageNumber < $this->navigation['total_pages'] ? $this->pageNumber + 1 : null;
}

public function gotoPage(int $pageNumber): void
{
    // Make sure the page number is within the bounds of the book
    if ($pageNumber >= 1 && $pageNumber <= $this->navigation['total_pages']) {
        $this->pageNumber = $pageNumber;
        $this->loadPage();
        $this->dispatch('url:replace', route('book.read', ['bookId' => $this->bookId, 'pageNumber' => $this->pageNumber]));
    }
}

public function render()
{
    return view('livewire.reader.book-reader', [
        'book' => $this->book,
        'toc' => $this->toc,
        'currentContent' => $this->currentContent,
        'pageNumber' => $this->pageNumber,
        'navigation' => $this->navigation,
    ]);
}
```
