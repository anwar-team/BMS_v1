    <script>
        function searchApp() {
            return {
                // Search state
                searchQuery: '',
                searchType: 'auto',
                currentPage: 1,
                loading: false,
                hasSearched: false,
                searchTimer: null,
                
                // Filters
                filters: {
                    author: '',
                    section: '',
                    book: ''
                },
                
                // Data
                results: {},
                authors: [],
                sections: [],
                books: [],
                
                // Initialize the component
                async init() {
                    await this.loadMetadata();
                    
                    // Check for URL params
                    const urlParams = new URLSearchParams(window.location.search);
                    if (urlParams.get('q')) {
                        this.searchQuery = urlParams.get('q');
                        this.performSearch();
                    }
                },
                
                // Load filter metadata
                async loadMetadata() {
                    try {
                        const response = await fetch('/search/metadata');
                        const data = await response.json();
                        this.authors = data.authors || [];
                        this.sections = data.sections || [];
                        this.books = data.books || [];
                    } catch (error) {
                        console.error('Failed to load metadata:', error);
                    }
                },
                
                // Debounced search (250ms delay)
                debouncedSearch() {
                    clearTimeout(this.searchTimer);
                    this.searchTimer = setTimeout(() => {
                        if (this.searchQuery.trim()) {
                            this.performSearch();
                        } else {
                            this.results = {};
                            this.hasSearched = false;
                        }
                    }, 250);
                },
                
                // Perform search
                async performSearch() {
                    if (!this.searchQuery.trim()) {
                        return;
                    }
                    
                    this.loading = true;
                    this.hasSearched = true;
                    
                    try {
                        const params = new URLSearchParams({
                            q: this.searchQuery,
                            search_type: this.searchType,
                            page: this.currentPage,
                            ...this.filters
                        });
                        
                        // Update URL
                        const newUrl = `${window.location.pathname}?${params.toString()}`;
                        history.pushState(null, '', newUrl);
                        
                        const response = await fetch(`/search/ajax?${params.toString()}`);
                        const data = await response.json();
                        
                        if (response.ok) {
                            this.results = data;
                        } else {
                            throw new Error(data.error || 'Search failed');
                        }
                    } catch (error) {
                        console.error('Search error:', error);
                        this.results = { results: [], total: 0, error: error.message };
                    } finally {
                        this.loading = false;
                    }
                },
                
                // Change page
                changePage(page) {
                    if (page && page !== this.currentPage) {
                        this.currentPage = page;
                        this.performSearch();
                    }
                },
                
                // Get search type label in Arabic
                getSearchTypeLabel(type) {
                    const labels = {
                        'exact_phrase': 'مطابقة تامة',
                        'phrase_match': 'عبارة كاملة', 
                        'multi_field': 'بحث شامل',
                        'fuzzy_fallback': 'بحث مرن',
                        'auto': 'تلقائي'
                    };
                    return labels[type] || type;
                }
            }
        }
    </script>
